import { Meta } from "@storybook/addon-docs/blocks";
import { Page, Rule, Rules, Section } from "./guide/Guide";

<Meta title="Foundations/Groups" />

<Page
  eyebrow="Groups"
  title="Fourteen groups, one question each."
  lede="The sidebar is being regrouped by the question a developer asks when they look for a component. This page is the target for that move: every public component, the group it goes to, and the reason where it could sit in two. Deprecated names are left out; they go in 0.12."
>
  <Section
    eyebrow="The rule"
    title="One folder per group."
    lede="Each group is one folder under src/components and one Storybook group of the same name. Sub-folders inside a group keep to about eight entries. A shared module (the field frame, the date arithmetic, the timeline scale) stays beside the components that use it."
  >
    <Rules>
      <Rule>A component goes where a developer looks for it first. Where it could sit in two groups, its docs say so under "when not to use", naming the other group.</Rule>
      <Rule>Foundations and Patterns stay sections of their own, outside src/components.</Rule>
      <Rule>The move happens once, in one commit, after every merge batch has landed. Export names do not change, so a product's imports stay the same.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Actions" title="Let someone do something.">
    <Rules>
      <Rule>`Button`, `Toolbar`, `Menu`, `CopyValue`.</Rule>
      <Rule>`Toolbar` stays here although tables use it most: it is one component for a list toolbar, a selection bar, unsaved changes, the end of a form and row actions. Tables documents the table roles and links back.</Rule>
      <Rule>`Menu` holds a row's actions (`role="menu"`). It does not move anyone anywhere, so it is not Navigation.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Forms" title="Let someone enter or choose.">
    <Rules>
      <Rule>Text: `TextField`, `TextArea`, `SearchField`, `QuantityField`, `FileField`.</Rule>
      <Rule>Select: `Select`, `ComboBox`, `MultiComboBox`.</Rule>
      <Rule>Choice: `ChoiceGroup`, `Checkbox`, `Radio`, `Switch`.</Rule>
      <Rule>Dates: `DateField`, `TimeField`, `DateRangePicker`, with the date arithmetic beside them.</Rule>
      <Rule>Field: `FieldLabel`, `FieldHint`, and the field frame with its `confidence`.</Rule>
      <Rule>`CreateFlow` is a stepped form in a page or a dialog. It renders `Steps` but owns the form state, so it is a form, not navigation or an overlay.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Tables" title="Show many rows to scan, sort and filter.">
    <Rules>
      <Rule>`DataTable`, `InfiniteTable`, `NewEntries`.</Rule>
      <Rule>`NewEntries` only appears inside `InfiniteTable`, so it moves here from Records.</Rule>
      <Rule>The table toolbar and the selection bar are `Toolbar` roles (Actions). The stories here show them in place.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Records" title="Show one record, or a short list of them.">
    <Rules>
      <Rule>`RecordCard`, `FactList`, `ProposalPanel`, `AttentionList`, `IconTile`.</Rule>
      <Rule>`ProposalPanel` is an inferred claim with its basis and, when it asks, its actions. It is a record a model wrote, so it sits with the records it is about.</Rule>
      <Rule>`IconTile` is the lead of a record's row or card and nothing else.</Rule>
      <Rule>A list beside its record is `Split variant="list"`, which stays in Layout because it arranges two panes. Its rows look like the selected row of a record list; Records links to it.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Figures" title="Show a number.">
    <Rules>
      <Rule>`Figure`, `FigureRow`, `Sparkline`.</Rule>
      <Rule>`Sparkline` has no axis and is read beside a figure, as a child of `Figure`. In Charts it would invite charts without an axis.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Text" title="Set text.">
    <Rules>
      <Rule>`Eyebrow`, `MetaLine`, `TextLink`, `Kbd`, `CodeBlock`, `Highlight`, `PullQuote`, `TickRule`, `Icon`.</Rule>
      <Rule>`TextLink` is a link in running text or at a block's edge. Moving between places is the job of the sidebar, the tabs and the breadcrumb, so it leaves Navigation.</Rule>
      <Rule>`Icon` sizes a glyph to the text it sits in.</Rule>
      <Rule>The page's and a section's heading is `PageHeader` (Layout), because it also holds the status, the meta line and the page's action.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Status" title="Show the state of a thing, inline.">
    <Rules>
      <Rule>`StatusPill`, `SeverityMark`, `SealValue`, `Badge`, `Tag`, `LevelMeter`.</Rule>
      <Rule>`LevelMeter` states one value on a scale (`role="meter"`). It has no axis and no series, so it is not a chart.</Rule>
      <Rule>`Tag` is a label that can be pressed or removed, which also makes it the filter chip and the entry token. It names a thing; `StatusPill` names a state from a closed list.</Rule>
      <Rule>`Badge` is a count on a control; a button draws it through `badge`.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Feedback" title="Tell someone what happened, or what is happening.">
    <Rules>
      <Rule>`Notice` (banner, empty, line), `Toast`, `Loading`, `Progress`, `RouteProgress`, `Delayed`.</Rule>
      <Rule>`Progress` is a counted wait (`role="progressbar"`). It tells the reader that work is under way, which is this group's question. A quota such as "12 of 25 seats" is a `LevelMeter` in Status.</Rule>
      <Rule>An empty list is `Notice variant="empty"`, and "everything is running" is `Notice variant="line"`.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Charts" title="Plot data on a scale.">
    <Rules>
      <Rule>`Chart`, `TrendChart`, `Seismograph`, `Distribution`, `CoverageMatrix`.</Rule>
      <Rule>A chart has an axis or a scale. A single value is a figure or a meter.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Timelines" title="Events and periods over time.">
    <Rules>
      <Rule>`EventTimeline`, `TimelineSummary`, `ChainTimeline`, `PeriodTimeline`, with the shared scale and types.</Rule>
      <Rule>They plot on a time axis like a chart, but each brings its own interaction: zoom and pan, episodes, links between lanes. A developer looking for "what happened when" looks here first.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Layout" title="Arrange a page.">
    <Rules>
      <Rule>Page: `Page`, `PageHeader`, `Panel`, `DetailSection`.</Rule>
      <Rule>Arrange: `Stack`, `Grid`, `Split`, `Disclosure`.</Rule>
      <Rule>`Panel` covers the danger zone (`tone="danger"`) and the hero (`surface="hero"`). `PageHeader` covers a section's heading (`level={2}`).</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Navigation" title="Move between places.">
    <Rules>
      <Rule>`Sidebar`, `Breadcrumb`, `Tabs`, `TabNav`, `Pagination`, `Steps`, `CommandBar`.</Rule>
      <Rule>`Sidebar` is the app's navigation. The frame around it is Shell, but a developer adding a section to the app looks for it here.</Rule>
      <Rule>`Steps` shows where a reader is in a bounded flow. It also lists the steps of a process on a brand page; that use is secondary.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Overlays" title="Show something above the page.">
    <Rules>
      <Rule>`Dialog` (side, centre, bottom), `ConfirmDialog`, `ShortcutHelp`, `Flyout`, `Tooltip`, `TooltipTarget`.</Rule>
      <Rule>`Tooltip` names an unlabelled control in a floating plane. It reports nothing, so it is not Feedback.</Rule>
    </Rules>
  </Section>

  <Section eyebrow="Shell" title="The app frame.">
    <Rules>
      <Rule>`AppShellBar` (with `AppShellBarEnd` and `AppShellBarKeep`), `AppShellContent`, `SignIn`, `SkipLink`, `ThemeProvider`, `DensityProvider`.</Rule>
      <Rule>The frame is a `Sidebar` (Navigation), then in `SidebarInset` the bar and the content. The bar is one row at every width; its status folds into "More" on a phone, and what is in `AppShellBarKeep`, the notification bell, stays.</Rule>
    </Rules>
  </Section>
</Page>
